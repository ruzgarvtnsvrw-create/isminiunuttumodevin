import java.util.Scanner;

public class Soru1 {

    public static void main(String[] args) {

        	Scanner ruzgar = new Scanner(System.in);

        			int n;
        			int i = 1;

       
 System.out.println("Bir N sayisi giriniz:");
        
n = ruzgar.nextInt();

        if (n > 0) {

            while (i <= n) {
               
 System.out.println(i);
               
 i = i + 1;
            }

        } else {
           
 System.out.println("Pozitif bir sayi girmeniz gerekiyor.");
        }


    }
}
