import java.util.Scanner;

public class Soru6 {

    public static void main(String[] args) {

        Scanner ruzgar = new Scanner(System.in);

        int sayi;
        
int ters = 0;
        		int kalan;

        

System.out.print("Ters cevrilecek sayiyi giriniz: ");
       	
 sayi = ruzgar.nextInt();

        int eskiSayi = sayi;

        
if (sayi < 0) {
            sayi = sayi * -1;
        }

        while (sayi > 0) {

            kalan = sayi % 10;
            ters = ters * 10 + kalan;
            sayi = sayi / 10;

        }

        if (eskiSayi < 0) {
            ters = ters * -1;
        }

        System.out.println("Sayinin tersi: " + ters);

    }
}
