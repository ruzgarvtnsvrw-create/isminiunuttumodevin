import java.util.Scanner;

public class Soru4 {

    public static void main(String[] args) {

        Scanner ruzgar = new Scanner(System.in);

        int sayi;
        	long faktoriyel = 1;

        	System.out.print("Faktoriyeli alinacak sayiyi giriniz: ");
        		sayi = ruzgar.nextInt();

        		if (sayi < 0) {

            System.out.println("Negatif sayilarin faktoriyeli olmaz.");

        } 	else if (sayi > 20) {

           
 System.out.println("Bu programda 20'den buyuk sayi girmeyiniz.");

        } else {

            
for (int i = 1; i <= sayi; i++) {

                faktoriyel = faktoriyel * i;

            }

            System.out.println("Sonuc = " + faktoriyel);

        }

    }






























}
